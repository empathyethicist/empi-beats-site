    |freq = 440, spread = 5, amp = 0.5, gate = 1|
    var env = EnvGen.ar(Env.perc(0.01, 0.5), gate, doneAction: 2);
    var pulses = Mix(Series.fill(5, { |i|
        Pulse.ar(freq + (i - 2.5) * spread, 0.3) * env
    }));
    var filtered = pulses * amp;
    Out.ar(0, filtered \! 2);
}).writeDefFile;
