    var env = EnvGen.ar(Env.perc(0.1, 10), gate, doneAction: 2);
    var sig = SinOsc.ar(freq, 0, 0.3) + SinOsc.ar(freq * 1.01, 0, 0.3);
    var filtered = LPF.ar(sig, 50);
    Out.ar(0, filtered * env * amp ! 2);
}).writeDefFile;
